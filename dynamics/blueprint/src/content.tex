\begin{abstract}
A shared library for finite stochastic dynamics using normalized real weights,
finite sums, and limits of finite-time probabilities. These are generic library
results, rather than statements attributed to a particular opinion-dynamics paper.
\end{abstract}

\section{Uniform expectations}
\begin{definition}[Finite uniform average]\label{def:avg}
\lean{Dynamics.avg}\leanok
For a finite type $A$, set $\operatorname{avg}(f)=\sum_{a\in A}f(a)/|A|$.
When $A$ is empty this is zero, following Lean's division convention.
\end{definition}
\begin{lemma}[Product independence]\label{lem:uniform-product}
\lean{Dynamics.avg_pi_prod,Dynamics.avg_pi_eval,Dynamics.avg_prod_pi,Dynamics.avg_equiv,Dynamics.avg_prod_eq_avg_avg,Dynamics.avg_pi_split}\leanok\uses{def:avg}
Uniform averages over a finite (dependent) product factor over its coordinates, each
coordinate is uniform on its factor, disjoint blocks of coordinates are independent,
and averages are invariant under bijections of the sample space.
\end{lemma}
\begin{lemma}[Finite trajectories]\label{lem:uniform-trajectory}
\lean{Dynamics.expList,Dynamics.expList_eq_avg_ofFn}\leanok\uses{def:avg}
The recursively defined expectation over $T$ independent draws equals the
uniform average over functions $\{0,\ldots,T-1\}\to A$.
\end{lemma}
\begin{proof}\leanok
Separate the first coordinate using the finite-function/product equivalence,
then induct on $T$. The empty sample space is handled separately.
\end{proof}
\begin{lemma}[Averages commute]\label{lem:avg-commute}
\lean{Dynamics.avg_avg_swap,Dynamics.expList_avg_comm,Dynamics.expList_zero_fun,Dynamics.expList_finset_sum}\leanok\uses{lem:uniform-trajectory}
Iterated uniform averages over two finite sets commute; in particular an outer
average commutes with the trajectory expectation $\mathrm{expList}$, which also
commutes with finite sums and vanishes on the zero functional.
\end{lemma}
\begin{proof}\leanok
Exchange the order of finite summation; for $\mathrm{expList}$, induct on $T$.
\end{proof}
\begin{lemma}[Prefixes of trajectories]\label{lem:expList-prefix}
\lean{Dynamics.expList_le_of_length,Dynamics.expList_congr_length,Dynamics.expList_take,Dynamics.expList_take_getElem?}\leanok\uses{lem:uniform-trajectory}
The expectation over $T$ draws only evaluates its functional on lists of length $T$. For
$k\le T$, a functional of the first $k$ draws has the same expectation over $T$ draws as over
$k$ draws; for $k<T$, the draw after them is a fresh uniform sample.
\end{lemma}
\begin{proof}\leanok
Induction on $T$; split $T=k+m$ and average out the last $m$ draws.
\end{proof}

\section{Weighted distributions and kernels}
\begin{definition}[Normalized real weights]\label{def:distribution}
\lean{Dynamics.Distribution,Dynamics.Distribution.expect,Dynamics.Distribution.prob}\leanok
A distribution consists of nonnegative real weights $p_a$ with $\sum_a p_a=1$.
Its expectation is $\sum_a p_a f(a)$, and an event probability is the expectation
of its indicator. Such a distribution cannot exist on an empty type.
\end{definition}
\begin{lemma}[Uniform agreement]\label{lem:uniform-agreement}
\lean{Dynamics.Distribution.uniform_expect}\leanok\uses{def:distribution,def:avg}
On a nonempty finite type the normalized uniform distribution agrees with the
existing finite average.
\end{lemma}
\begin{lemma}[Independent products and pushforward]\label{lem:weighted-product}
\lean{Dynamics.Distribution.independent_expect_prod,Dynamics.Distribution.independent_expect_eval,Dynamics.Distribution.map_expect}\leanok\uses{def:distribution}
Independent coordinates have product weights. Expectations of product
observables factor, and transporting weights along a function commutes with expectation.
\end{lemma}
\begin{proof}\leanok
Distribute products over finite sums; for pushforward interchange the two sums
and evaluate each fiber indicator.
\end{proof}
\begin{lemma}[Independent coordinates]\label{lem:independent}
\lean{Dynamics.Distribution.independent_expect_update,Dynamics.Distribution.independent_expect_eq_sum_update,Dynamics.Distribution.independent_expect_fin_succ,Dynamics.Distribution.independent_prob_fin_succ,Dynamics.Distribution.independent_expect_mul_of_update,Dynamics.Distribution.independent_expect_pair,Dynamics.Distribution.independent_expect_sum_sq,Dynamics.Distribution.independent_expect_sum_cube,Dynamics.Distribution.expect_expect_coord,Dynamics.Distribution.expect_mul_of_dep}\leanok\uses{lem:weighted-product}
Under an independent product, resampling one coordinate does not change the law, so the
expectation can be conditioned on the value of one coordinate. Two distinct coordinates are
independent, and so are functions of disjoint sets of coordinates; centred coordinate functions
have additive second and third moments. If $x$ and $y$ are independent families, the family
$(\varphi_i(x_i,y_i))_i$ is independent, with the coordinatewise laws.
\end{lemma}
\begin{proof}\leanok
Resampling: the bijection $(x,a)\mapsto(x[i\mapsto a],x_i)$ moves the weight of coordinate $i$.
Moments: expand the power of the sum and factor the mixed terms. Combination: expand the product
weights of $(\varphi_i(x_i,y_i))_i$ as sums over the preimages.
\end{proof}
\begin{lemma}[Bridges between the expectation APIs]\label{lem:bridge}
\lean{Dynamics.avg_eq_expect,Dynamics.expList_eq_expect,Dynamics.Distribution.independent_uniform_expect,Dynamics.expList_eq_independent_expect}\leanok\uses{lem:uniform-trajectory,lem:uniform-agreement,lem:weighted-product}
The finite average is Mathlib's finite expectation, $\operatorname{avg}(f)=\E_{a\in A}f(a)$
(\texttt{Finset.expect}); both vanish on an empty type. On a nonempty type the independent
product of uniform distributions on $A$, indexed by a finite set $I$, is the uniform
distribution on $A^I$. Hence $\mathrm{expList}$ over $T$ rounds is the expectation of
$F$ over $T$ i.i.d.\ uniform draws, both as Mathlib's $\E_{\omega\colon\{0,\ldots,T-1\}\to A}$
and under the product distribution.
\end{lemma}
\begin{proof}\leanok
Mathlib's finite expectation is a sum divided by a cardinality. Every point of $A^I$ has
product weight $|A|^{-|I|}=|A^I|^{-1}$. Combine both with Lemma~\ref{lem:uniform-trajectory}.
\end{proof}
\begin{lemma}[Bridge to Mathlib's probability mass functions and measures]\label{lem:bridge-pmf}
\lean{Dynamics.Distribution.toPMF,Dynamics.Distribution.toPMF_uniform,Dynamics.Distribution.toPMF_point,Dynamics.Distribution.toPMF_map,Dynamics.Distribution.expect_eq_integral,Dynamics.Distribution.prob_eq_toMeasure,Dynamics.Distribution.toMeasure_independent,Dynamics.Distribution.iIndepFun_independent}\leanok\uses{def:distribution,lem:weighted-product}
A distribution $p$ on a finite type is a probability mass function (Mathlib's \texttt{PMF}); the
uniform distribution, point masses and pushforward are \texttt{PMF.uniformOfFintype},
\texttt{PMF.pure} and \texttt{PMF.map}. Equip the type with a $\sigma$-algebra containing the
singletons (hence every set). Under the induced measure $\mu_p$, the expectation of $f$ is
$\int f\,d\mu_p$ and the probability of an event $S$ is $\mu_p(S)$. The measure of an independent
product is the product measure $\bigotimes_i\mu_{p_i}$, so the coordinates
$x\mapsto Y_i(x_i)$ are independent random variables in Mathlib's sense (\texttt{iIndepFun}).
\end{lemma}
\begin{proof}\leanok
All integrals are finite sums of weights. Two measures on a finite type agree when they agree on
singletons, and a singleton of the product space has product weight on both sides; Mathlib's
\texttt{iIndepFun\_pi} gives independence under the product measure.
\end{proof}
\begin{definition}[Transition kernel]\label{def:kernel}
\lean{Dynamics.Kernel,Dynamics.Kernel.iterate,Dynamics.Kernel.trajectory}\leanok\uses{def:distribution}
A finite kernel assigns a distribution to every state. Its operator iterates
compute expectations at finite times; trajectory functionals additionally
record the history.
\end{definition}
\begin{lemma}[Transition matrix]\label{lem:kernel-matrix}
\lean{Dynamics.Kernel.toMatrix,Dynamics.Kernel.apply_eq_toMatrix_mulVec,Dynamics.Kernel.iterate_eq_toMatrix_pow_mulVec,Dynamics.Kernel.toMatrix_mem_rowStochastic}\leanok\uses{def:kernel}
The matrix $P_{ab}=K(a)(b)$ of a kernel is row stochastic, the transition operator is $f\mapsto Pf$
and the expectation after $n$ steps is $P^nf$.
\end{lemma}
\begin{proof}\leanok
Unfold the expectation as a sum; induction on $n$.
\end{proof}
\begin{lemma}[Endpoint agreement]\label{lem:endpoint}
\lean{Dynamics.Kernel.trajectory_endpoint}\leanok\uses{def:kernel}
Ignoring history in a trajectory expectation recovers the iterated transition operator.
\end{lemma}

\section{Stationarity and absorption}
\begin{theorem}[Existence of stationary weights]\label{thm:stationary}
\lean{Dynamics.Kernel.exists_stationary}\leanok\uses{def:kernel}
Every stochastic kernel on a nonempty finite type has a distribution $p$ with $pK=p$.
\end{theorem}
\begin{proof}\leanok
Average the first $n+1$ evolved distributions. The stationarity defect of this
Ces\`aro average is $(p_{n+1}-p_0)/(n+1)$, which tends to zero coordinatewise.
Compactness of the finite probability simplex gives a convergent subsequence;
continuity of finite sums implies that its limit is stationary.
\end{proof}
\begin{lemma}[Uniform absorption block]\label{lem:block}
\lean{Dynamics.Kernel.exists_uniform_block}\leanok\uses{def:kernel}
Let $f$ be a zero-one survival indicator with $Kf\le f$. Suppose for each
state $a$ there is $n_a$ with $(K^{n_a}f)(a)<1$. There are $m>0$ and
$0\le q<1$ such that $K^m f\le qf$.
\end{lemma}
\begin{proof}\leanok
Take one plus the maximum of the finitely many access times. Antitonicity of
survival probabilities puts every state below one at that time. Their finite
maximum supplies $q$. At states with $f=0$, absorption keeps the expectation zero.
\end{proof}
\begin{lemma}[Access from reachability]\label{lem:reachable-access}
\lean{Dynamics.Kernel.iterate_succ_lt_one,Dynamics.Kernel.exists_iterate_lt_one_of_reachable}\leanok\uses{def:kernel}
Let $f\le1$. If $t_0$ is reachable from $s$ along transitions $a\to b$ with
$K(a,b)>0$ and $f(t_0)=0$, then $(K^n f)(s)<1$ for some $n$.
\end{lemma}
\begin{proof}\leanok
Induct on the chain. If $K(s,t)>0$ and $(K^n f)(t)<1$, then $(K^{n+1}f)(s)$ is an average of
values at most one with weight $K(s,t)>0$ on a value below one.
\end{proof}
\begin{theorem}[Geometric decay and convergence]\label{thm:absorption}
\lean{Dynamics.Kernel.geometric_blocks,Dynamics.Kernel.finite_absorption}\leanok\uses{lem:block}
Under these hypotheses, $K^{nm}f\le q^n f$, and $K^t f(a)\to0$ for every state $a$.
\end{theorem}
\begin{proof}\leanok
Induct over blocks using positivity and linearity. The geometric bound tends
to zero. For arbitrary $t$, compare with $m\lfloor t/m\rfloor$ by antitonicity.
\end{proof}
\begin{theorem}[Finite-horizon optional stopping]\label{thm:optional-stopping}
\lean{Dynamics.Kernel.event_eq_iterate,Dynamics.Kernel.eq_add_target_add_survival,Dynamics.Kernel.iterate_eq_event_add,Dynamics.Kernel.event_error_of_invariant,Dynamics.Kernel.tendsto_event_of_invariant,Dynamics.Kernel.iSup_event_of_invariant}\leanok\uses{def:kernel}
Let $\varphi$ equal $\varphi_A$ on an event $A$ and $\varphi_B$ on an event $B$, and let
$S$ be the event of lying in neither. If $\E_{x_0}\varphi(X_T)=\varphi(x_0)$ and
$m\le\varphi-\varphi_B\le M$ on $S$, then
$\varphi(x_0)-\varphi_B-(\varphi_A-\varphi_B)\Pr_{x_0}(X_T\in A)$ lies between
$m\Pr_{x_0}(X_T\in S)$ and $M\Pr_{x_0}(X_T\in S)$. Hence, if $\varphi_A\ne\varphi_B$,
the conservation law holds for every $T$ and $\Pr_{x_0}(X_T\in S)\to0$, then
$\Pr_{x_0}(X_T\in A)\to(\varphi(x_0)-\varphi_B)/(\varphi_A-\varphi_B)$; if moreover $A$
is absorbing, this is the supremum of the finite-time probabilities.
\end{theorem}
\begin{proof}\leanok
Pointwise, $\varphi-\varphi_B=(\varphi_A-\varphi_B)1_A+(\varphi-\varphi_B)1_S$ (on
$A\cap B$ both sides vanish). Take expectations at time $T$ and bound the last term by
monotonicity. For the limit, take $M=-m=\sum_a|\varphi(a)-\varphi_B|$ and squeeze. If $A$
is absorbing, the finite-time probabilities increase, so their limit is their supremum.
\end{proof}
\begin{lemma}[Gambler's ruin for two consensus configurations]\label{lem:gamblers-ruin}
\lean{Dynamics.Kernel.harmonic_affine,Dynamics.Kernel.iSup_consensus_of_harmonic,Dynamics.Kernel.iSup_consensus_of_invariant}\leanok\uses{thm:optional-stopping}
Let $K$ be a kernel on configurations $V\to\{0,1\}$ for which the all-one configuration
$\mathbf 1$ is absorbing and the non-consensus probability tends to zero from $s$. If
$K\varphi=\varphi$ and $\varphi(\mathbf 1)\ne\varphi(\mathbf 0)$, the probability of fixation
in $\mathbf 1$ from $s$ is $(\varphi(s)-\varphi(\mathbf 0))/(\varphi(\mathbf 1)-\varphi(\mathbf 0))$.
For instance, a harmonic $q^{k}$, with $k$ the number of ones out of $n$, gives
$(1-q^k)/(1-q^n)$. Affine images $u\varphi+v$ of harmonic observables are harmonic.
\end{lemma}
\begin{proof}\leanok
Apply the previous theorem with $A=\{\mathbf 1\}$, $B=\{\mathbf 0\}$; the event of lying in
neither is non-consensus. Affine images are harmonic by linearity of expectation.
\end{proof}

\section{Rounds, concentration and phases}
\begin{definition}[Kernel of a round-based process]\label{def:ofstep}
\lean{Dynamics.Kernel.ofStep,Dynamics.Kernel.iterate_ofStep,Dynamics.Kernel.event_ofStep}\leanok\uses{def:kernel}
A deterministic update $\mathrm{step}\colon S\times R\to S$ driven by a uniform
round $r\in R$ defines the kernel $K(s)=\mathrm{law}(\mathrm{step}(s,r))$. Its
$T$-step expectations are expectations over $T$ i.i.d.\ uniform rounds.
\end{definition}
\begin{proof}\leanok
Induction on $T$, unfolding one round of $\mathrm{expList}$.
\end{proof}
\begin{theorem}[Time reversal of i.i.d.\ rounds]\label{thm:reverse}
\lean{Dynamics.reverse_ofFn,Dynamics.expList_comp_reverse,Dynamics.Kernel.iterate_ofStep_foldr}\leanok\uses{lem:uniform-trajectory,def:ofstep}
For every functional $F$ of $T$ i.i.d.\ uniform rounds,
$\E F(r_T,\dots,r_1)=\E F(r_1,\dots,r_T)$. In particular the $T$-step expectations of a
round-based process are also obtained by applying the rounds last to first.
\end{theorem}
\begin{proof}\leanok
Write both sides as uniform averages over $\mathrm{Fin}\,T\to R$ and reindex by the
involution $i\mapsto T-1-i$.
\end{proof}
\begin{theorem}[Hoeffding and Bernstein inequalities]\label{thm:concentration}
\lean{Dynamics.avg_hoeffding,Dynamics.avg_bernstein}\leanok\uses{def:avg}
Let $X=\sum_{i<n}Y_i(\omega_i)$ with independent uniform coordinates. If each
$Y_i$ is $\{0,1\}$-valued, $\Pr(X\ge\E X+\lambda)\le e^{-2\lambda^2/n}$. If
$Y_i-\E Y_i\le b$ and $\sum_i\mathrm{Var}\,Y_i\le\sigma^2$, then
$\Pr(X\ge\E X+\lambda)\le\exp\bigl(-\lambda^2/(2\sigma^2(1+b\lambda/(3\sigma^2)))\bigr)$.
\end{theorem}
\begin{proof}\leanok
Markov's inequality applied to $e^{tX}$, which factors over the coordinates.
Hoeffding's lemma bounds each Bernoulli factor; for Bernstein,
$e^y\le1+y+y^2/(2(1-u/3))$ for $y\le u<3$ bounds each factor, and $t$ is
chosen as $\lambda/(\sigma^2+b\lambda/3)$.
\end{proof}
\begin{theorem}[Hoeffding's inequality, derived from Mathlib]\label{thm:hoeffding-pmf}
\lean{Dynamics.Distribution.hoeffding}\leanok\uses{lem:bridge-pmf}
Let $X=\sum_{i\in I}Y_i(x_i)$ under the independent product of distributions $p_i$, where
$Y_i$ takes values in $[a_i,b_i]$. Then
$\Pr(X\ge\sum_i\E_{p_i}Y_i+\lambda)\le\exp\bigl(-2\lambda^2/\sum_i(b_i-a_i)^2\bigr)$ for
$\lambda\ge0$. The case of uniform $p_i$ and $\{0,1\}$-valued $Y_i$ is the Hoeffding bound of
Theorem~\ref{thm:concentration}.
\end{theorem}
\begin{proof}\leanok
Transport the statement to the induced measure by Lemma~\ref{lem:bridge-pmf}. Each centred
coordinate is sub-Gaussian with parameter $(b_i-a_i)^2/4$ by Mathlib's Hoeffding lemma, and the
coordinates are independent, so Mathlib's Hoeffding inequality for sums of independent
sub-Gaussian variables applies.
\end{proof}
\begin{theorem}[Chernoff bounds for independent Bernoulli trials]\label{thm:chernoff}
\lean{Dynamics.Distribution.bernoulli,Dynamics.Distribution.chernoff_upper_ratio,Dynamics.Distribution.chernoff_upper,Dynamics.Distribution.chernoff_lower_ratio,Dynamics.Distribution.chernoff_lower,Dynamics.Distribution.bernoulli_chernoff_upper_ratio,Dynamics.Distribution.bernoulli_chernoff_upper,Dynamics.Distribution.bernoulli_chernoff_lower_ratio,Dynamics.Distribution.bernoulli_chernoff_lower,Dynamics.Distribution.independent_bernoulli_prob_split,Dynamics.Distribution.binTail,Dynamics.Distribution.binTail_succ_succ,Dynamics.avg_chernoff_upper_ratio,Dynamics.avg_chernoff_upper,Dynamics.avg_chernoff_lower_ratio,Dynamics.avg_chernoff_lower}\leanok\uses{def:distribution,def:avg}
Let $X=\sum_iX_i$ be a sum of independent $\{0,1\}$-valued trials with
$\Pr(X_i=1)=p_i$, and let $\mu_L\le\mu=\sum_ip_i\le\mu_H$. For $\delta>0$,
\[\Pr(X\ge(1+\delta)\mu_H)\le\Bigl(\frac{e^{\delta}}{(1+\delta)^{1+\delta}}\Bigr)^{\mu_H}
\le e^{-\delta^2\mu_H/(2+\delta)},\]
and for $0<\delta<1$,
\[\Pr(X\le(1-\delta)\mu_L)\le\Bigl(\frac{e^{-\delta}}{(1-\delta)^{1-\delta}}\Bigr)^{\mu_L}
\le e^{-\delta^2\mu_L/2}.\]
The trials are $\{0,1\}$-valued observables of the coordinates of an independent
product of distributions, independent biased coins (counting the heads in any
subset), or the coordinates of one uniform round (Mitzenmacher--Upfal,
\emph{Probability and Computing}, Theorems 4.4 and 4.5 with Exercise 4.7).
For identical Bernoulli$(p)$ coins, conditioning on one coin gives
$\Pr(E)=p\,\Pr(E\mid\text{heads})+(1-p)\Pr(E\mid\text{tails})$, and the binomial tail
$T(m,k)=\Pr(\text{at least $k$ heads among $m$ coins})$ satisfies
$T(m+1,k+1)=p\,T(m,k)+(1-p)\,T(m,k+1)$.
\end{theorem}
\begin{proof}\leanok
Markov's inequality applied to $e^{tX}$, which factors over the coordinates; each
factor is $1+p_i(e^t-1)\le e^{p_i(e^t-1)}$, so $\E e^{tX}\le e^{\mu_H(e^t-1)}$ for
$t\ge0$ and $\E e^{tX}\le e^{\mu_L(e^t-1)}$ for $t\le0$. Take $t=\log(1+\delta)$,
respectively $t=\log(1-\delta)$. The closed forms follow from
$\log(1+\delta)\ge2\delta/(2+\delta)$ (the first term of the series of
$\log\frac{1+x}{1-x}$ at $x=\delta/(2+\delta)$) and
$(1-\delta)\log(1-\delta)\ge-\delta+\delta^2/2$. The conditioning formula sums over the
value of the coin (independence of the coordinates), and the recursion of $T$ is the case of
the first coin.
\end{proof}
\begin{theorem}[Lower tails, MGF and parametric Chernoff bounds]\label{thm:chernoffmgf}
\lean{Dynamics.avg_lower_tail_le_of_mgf,Dynamics.avg_hoeffding_lower,Dynamics.variance_le_avg_of_zero_one,Dynamics.avg_chernoff_mgf,Dynamics.avg_chernoff_upper_of_mgf,Dynamics.avg_chernoff_lower_of_mgf,Dynamics.avg_chernoff_upper_log,Dynamics.avg_chernoff_lower_log,Dynamics.avg_chernoff_lower_mul,Dynamics.avg_bernstein_lower,Dynamics.avg_bernstein_of_zero_one,Dynamics.avg_bernstein_lower_of_zero_one,Dynamics.Distribution.prob_ge_le_exp_tilt}\leanok\uses{thm:concentration,thm:chernoff}
Let $X=\sum_{i<n}Y_i(\omega_i)$ with independent uniform $\{0,1\}$-valued
coordinates and $\mu=\E X$. Then $\Pr(X\le\E X-\lambda)\le e^{-2\lambda^2/n}$, and a
$\{0,1\}$ coordinate has variance at most its mean. For every real $t$,
$\E e^{tX}\le\exp(\mu(e^t-1))$; hence $\Pr(X\ge k)\le\exp(\mu(e^t-1)-tk)$ for
$t\ge0$ and $\Pr(X\le k)\le\exp(\mu(e^t-1)-tk)$ for $t\le0$. If $\E X\le\mu\le k$
with $\mu>0$, then $\Pr(X\ge k)\le\exp(k-\mu-k\log(k/\mu))$; if $0<k\le\mu\le\E X$,
then $\Pr(X\le k)\le\exp(k-\mu-k\log(k/\mu))$. Finally
$\Pr(X\le(1-\delta)\E X)\le\exp(-\delta^2\E X/2)$ for $0\le\delta<1$.
Bernstein's bound also holds for the lower tail $\Pr(X\le\E X-\lambda)$ (when
$\E Y_i-Y_i\le b$); for $\{0,1\}$-valued coordinates and $\ell>0$, each of
$\Pr(X\ge\mu+\sqrt{3\ell(\mu+2\ell)})$ and $\Pr(X\le\mu-\sqrt{3\ell(\mu+2\ell)})$ is at
most $e^{-\ell}$. For independent (not necessarily uniform) $\{0,1\}$ trials with
$\E X\le(1-\varepsilon)c$, $0\le\varepsilon<1$ and $c\ge0$,
$\Pr(X\ge k)\le\exp(c(\varepsilon-\varepsilon^2/2)-\varepsilon k)$ for every $k$.
\end{theorem}
\begin{proof}\leanok
The lower Hoeffding tail is the upper one for $1-Y_i$. For the MGF bound,
$e^{tY}=1+(e^t-1)Y$ on $\{0,1\}$ and $1+y\le e^y$, factor by factor; Markov's
inequality applied to $e^{tX}$ gives the tails, and $t=\log(k/\mu)$ the closed forms
(the exponent is monotone in $\mu$ in the right direction). The last bound is the
lower bound of Theorem~\ref{thm:chernoff} with $\mu_L=\E X$ (trivial for $\delta=0$).
The lower Bernstein tail is the upper one for $-Y_i$; the $\{0,1\}$ forms take $b=1$,
$\sigma^2=\mu+2\ell$ and $\lambda=\sqrt{3\ell\sigma^2}$. The last bound is Markov's
inequality on $e^{\varepsilon X}$ with $(1-\varepsilon)e^{\varepsilon}\le1-\varepsilon^2/2$.
\end{proof}
\begin{theorem}[Moments and anti-concentration]\label{thm:moments}
\lean{Dynamics.avg_cauchy,Dynamics.avg_sum_sq,Dynamics.avg_sum_fourth_le,Dynamics.avg_pz_one_sided,Dynamics.avg_pz,Dynamics.avg_pz_sum,Dynamics.variance_of_zero_one,Dynamics.avg_pz_zero_one,Dynamics.avg_exp_sum_le,Dynamics.avg_sq_sum_le_of_pairwise}\leanok\uses{lem:uniform-product,lem:bridge}
Let $S=\sum_{i<n}f_i(\omega_i)$ with independent uniform coordinates, each $f_i$
centered and bounded by $1$ in absolute value, and $\sigma^2=\sum_i\E f_i^2$. Then
$\E S^2=\sigma^2$ and $\E S^4\le\sigma^2+3\sigma^4$. If $\E X\ge0$ and $0\le\theta\le1$,
then $(1-\theta)^2(\E X)^2\le\E X^2\,\Pr(X>\theta\,\E X)$ (one-sided Paley--Zygmund). If $\E Z^2=s>0$ and
$\E Z^4\le4s^2$, then $\Pr(s\le4Z^2)\ge9/64$ (two-sided Paley--Zygmund); hence
$\Pr(\sigma^2\le4S^2)\ge9/64$ whenever $\sigma^2\ge1$, in particular for centered
$\{0,1\}$-valued coordinates, whose variance is $p(1-p)$. For $\{0,1\}$-valued
coordinates and every real $t$, $\E e^{tS}\le\exp(t\,\E S+nt^2/8)$. Finally, a sum
$X$ of pairwise uncorrelated $\{0,1\}$-valued variables with mean $\mu$ has
$\E X^2\le\mu+\mu^2$.
\end{theorem}
\begin{proof}\leanok
The moments by induction on $n$, splitting off the first coordinate, which is
independent of the others. For Paley--Zygmund, $X\le\theta\,\E X+X\,1_A$ with
$A=\{X>\theta\,\E X\}$ pointwise, and Cauchy--Schwarz (Mathlib's, through
$\operatorname{avg}=\E$) gives $(\E[X 1_A])^2\le\E X^2\,\Pr(A)$; the two-sided form is the
one-sided one for $X=Z^2$ and $\theta=1/4$. The exponential moment factors over the
coordinates, and Hoeffding's lemma bounds each factor. For the last bound, expand
$X^2$: the diagonal terms are $\E X_i^2=\E X_i$ and the others $\E X_i\,\E X_j$.
\end{proof}
\begin{lemma}[Elementary tail bounds and monotonicity]\label{lem:tail}
\lean{Dynamics.Distribution.prob_mono,Dynamics.Distribution.expect_lt_one,Dynamics.avg_lt_one,Dynamics.avg_markov,Dynamics.avg_markov_one,Dynamics.Kernel.one_sub_avg_le_prob_ofStep,Dynamics.Kernel.iterate_monotone,Dynamics.Kernel.event_monotone,Dynamics.exp_neg_two_log,Dynamics.exp_neg_three_log,Dynamics.one_le_of_log_pos}\leanok\uses{def:distribution,def:ofstep}
Probability is monotone in the event. An observable $f\le1$ with $f(b)<1$ at a
point of positive weight has expectation $<1$. Markov's inequality
$\Pr(X\ge c)\le\E X/c$ for $X\ge0$, $c>0$; in particular
$\Pr(\sum_iY_i(\omega_i)\ge1)\le\sum_i\E Y_i$ for nonnegative coordinates. One
round of $\mathrm{step}$ lands in an event with probability at least
$1-\E[\mathrm{bad}]$ when $\mathrm{bad}\ge0$ is at least $1$ on every miss. If
$f\le Kf$, then $n\mapsto K^nf(a)$ is nondecreasing; in particular the occupation
probability of an absorbing event is nondecreasing in time. Finally
$e^{-2\log n}=1/n^2$ for $n\ge1$, $e^{-3\log n}=1/n^3$ for $n>0$, and $\log n>0$
forces $n\ge1$.
\end{lemma}
\begin{proof}\leanok
Pointwise comparisons of indicators, integrated by monotonicity of the expectation;
for the kernel statements, induction on time with $K^{n+1}f=K^n(Kf)$.
\end{proof}
\begin{lemma}[Scalar facts for round counts]\label{lem:scalar}
\lean{Dynamics.ceil_add_ceil_le,Dynamics.mul_ceil_le_ceil,Dynamics.nat_ceil_mul_le_of_le,Dynamics.le_pow_ceil_real,Dynamics.le_pow_ceil,Dynamics.exists_lt_pow_mul,Dynamics.exists_le_two_pow_two_pow,Dynamics.pow_three_le_of_log,Dynamics.log_cube_le,Dynamics.log_pow_four_div_le,Dynamics.half_le_log_one_add,Dynamics.mul_one_div_sq_le}\leanok
For $a,b\ge0$: $\lceil a\rceil+\lceil b\rceil\le\lceil c\rceil$ if $a+b+2\le c$, and
$k\lceil a\rceil\le\lceil b\rceil$ if $k(a+1)\le b$; $\lceil ay\rceil\le\lceil by\rceil$ if
$0\le a\le b$. For $w\ge1$ and $q>1$,
$w\le q^{\lceil\log w/\log q\rceil}$; growth by a factor $q$ from $\Delta>0$ passes
$N\ge\Delta$ within $\log(N/\Delta)/\log q+2$ steps, and $2^{2^T}\ge N\ge2$ for some
$T\le2\log\log N+3$. Moreover $c^3\le x$ if $3c\le\log x$; $(\log x)^3/6\le x$ and
$(\log x)^4/24\le x$ if $\log x\ge0$; $\varepsilon/2\le\log(1+\varepsilon)$ on $[0,1]$;
and $S/N^2\le C/N$ if $S\le CN$, $N>0$.
\end{lemma}
\begin{proof}\leanok
$\lceil a\rceil<a+1$; $w\le q^t$ iff $\log w\le t\log q$; $y^k/k!\le e^y$ at $y=\log x$; and
$1-1/x\le\log x$.
\end{proof}
\begin{theorem}[Nested phases]\label{thm:phases}
\lean{Dynamics.Kernel.nested_phases}\leanok\uses{def:kernel}
Let $A_1\supseteq\cdots\supseteq A_T$ be sets of states such that from $A_i$
the chain stays in $A_i$ with probability at least $1-\varepsilon$ and, for
$i<T$, enters $A_{i+1}$ with probability at least $1-\nu$. From $A_1$, after
$\ell T$ steps the chain is in $A_T$ with probability at least
$1-T(\ell\varepsilon+\nu^\ell)$.
\end{theorem}
\begin{proof}\leanok
Track the indicator of lying outside a set. In $\ell$ steps the chain leaves
$A_i$ with probability at most $\ell\varepsilon$, and misses $A_{i+1}$ in all
$\ell$ steps with probability at most $\nu^\ell$. Sum over the $T$ phases.
\end{proof}
\section{Drift and graph rounds}
\begin{definition}[Time-dependent chain]\label{def:iterateseq}
\lean{Dynamics.Kernel.iterateSeq,Dynamics.Kernel.iterateSeq_const}\leanok\uses{def:kernel}
A family of kernels $(K_t)_{t\ge0}$ moves the chain from time $t$ to time
$t+1$ with $K_t$; $\mathrm{iterateSeq}(K,n,f)(a)$ is the expectation of $f$ at
time $n$ from $a$. A constant family gives the iterates of a single kernel.
\end{definition}
\begin{proof}\leanok
Induction on $n$.
\end{proof}
\begin{theorem}[Drift lemma; Berenbrink et al., ICALP 2016, Lemma 2.2]\label{thm:drift}
\lean{Dynamics.Kernel.drift_absorption_seq,Dynamics.Kernel.drift_absorption}\leanok
\uses{def:iterateseq}
Let $\Psi\ge0$, and for $t<T$ let $c_t\ge0$,
$\E[\Psi(X_{t+1})\mid X_t=x]\le\Psi(x)-c_t/\Psi(x)$ when $\Psi(x)>0$, and
$\E[\Psi(X_{t+1})\mid X_t=x]=0$ when $\Psi(x)=0$. If
$\sum_{t<T}c_t\ge4\Psi(x_0)^2$, then $\Pr_{x_0}(\Psi(X_T)=0)\ge1/2$.
\end{theorem}
\begin{proof}\leanok
Let $q=\Pr(\Psi(X_T)>0)$; survival is nonincreasing, so it is at least $q$
before $T$. With $\lambda=q/\Psi(x_0)$, the bound $1/y\ge2\lambda-\lambda^2y$
linearizes the drift: $\E\Psi(X_{t+1})\le(1+c_t\lambda^2)\E\Psi(X_t)
-2c_t\lambda\Pr(\Psi(X_t)>0)\le\E\Psi(X_t)-c_t\lambda q$. (The paper uses
Jensen's inequality here; this is its optimal-$\lambda$ form.) Summing,
$0<m q\le\E\Psi(X_T)\le\Psi(x_0)(1-4q^2)$ if $q>0$, where $m>0$ is the least
positive value of $\Psi$; hence $q\le1/2$.
\end{proof}
\begin{theorem}[Multiplicative drift; the argument of Lemma 2.4 there]\label{thm:multdrift}
\lean{Dynamics.Kernel.multiplicative_drift_seq,Dynamics.Kernel.multiplicative_drift}\leanok
\uses{def:iterateseq}
Let $\Psi\ge0$ with every positive value at least $\Psi_{\min}>0$, and
$\E[\Psi(X_{t+1})\mid X_t=x]\le(1-\delta_t)\Psi(x)$ for all $x$ and $t<T$.
Then $\Pr_{x_0}(\Psi(X_T)>0)\le\prod_{t<T}(1-\delta_t)\,\Psi(x_0)/\Psi_{\min}$.
\end{theorem}
\begin{proof}\leanok
By induction $\E\Psi(X_T)\le\prod_{t<T}(1-\delta_t)\Psi(x_0)$ (the factors are
nonnegative unless $\Psi\equiv0$); then Markov's inequality
$\mathbf 1_{\Psi>0}\le\Psi/\Psi_{\min}$.
\end{proof}
\begin{lemma}[Synchronous neighbour rounds]\label{lem:neighborround}
\lean{Dynamics.NeighborRound,Dynamics.avg_neighborSet,Dynamics.neighborRound_nonempty,Dynamics.card_neighborRound,Dynamics.avg_neighborRound_prod,Dynamics.avg_neighborRound_eval,Dynamics.avg_neighborRound_mul,Dynamics.avg_neighborRound_sum,Dynamics.avg_vertex_neighborRound}\leanok
\uses{def:avg}
In a round $r\in\prod_vN(v)$ every vertex picks a neighbour. Under the uniform
distribution the picks are independent and uniform:
$\E\prod_vf_v(r_v)=\prod_v\E_{u\in N(v)}f_v(u)$, and, without isolated
vertices, $\E g(r_v)=\E_{u\in N(v)}g(u)$. A uniform vertex $v$ with a uniform
round gives the sequential step ``uniform vertex, uniform neighbour $r_v$''.
\end{lemma}
\begin{proof}\leanok
$\sum_r\prod_vf_v(r_v)=\prod_v\sum_{u\in N(v)}f_v(u)$ and
$|\prod_vN(v)|=\prod_v\deg v$; the marginals take $f_w=1$ for $w\ne v$.
\end{proof}
\begin{lemma}[Uniform oriented edge]\label{lem:edgeround}
\lean{Dynamics.EdgeRound,Dynamics.edgeRound_nonempty,Dynamics.avg_edgeRound,Dynamics.avg_edgeRound_symm,Dynamics.avg_edgeRound_edge,Dynamics.avg_edgeRound_fst}\leanok
\uses{def:avg}
For a uniform dart $d$ of $G$,
$\E F(d_1,d_2)=\sum_v\sum_{u\in N(v)}F(v,u)/(2|E|)$; reversing $d$ preserves
the law, its underlying edge is a uniform edge, and its tail is $v$ with
probability $\deg v/(2|E|)$.
\end{lemma}
\begin{proof}\leanok
Group the darts by tail (the darts with tail $v$ are the neighbours of $v$)
or by edge (each edge carries two darts).
\end{proof}
\begin{lemma}[Sums over darts]\label{lem:dartsums}
\lean{Dynamics.dartEquiv,Dynamics.sum_dart,Dynamics.walk_sum_darts,Dynamics.sum_darts_sq_le}\leanok
A sum over the darts of $G$ is the sum over the adjacent ordered pairs; along a walk from
$u$ to $v$ the differences $\psi(d_1)-\psi(d_2)$ telescope to $\psi(u)-\psi(v)$; and over
any set of distinct darts $\sum_d(\psi(d_1)-\psi(d_2))^2\le\sum_{i\sim j}(\psi(i)-\psi(j))^2$.
\end{lemma}
\begin{proof}\leanok
Darts are the adjacent ordered pairs; induction on the walk; distinct darts are distinct
ordered pairs and the summands are nonnegative.
\end{proof}
\begin{definition}[Lazy kernels and random walks on graphs]\label{def:randomwalk}
\lean{Dynamics.Kernel.lazy,Dynamics.Kernel.lazy_weight,Dynamics.Kernel.lazy_expect,Dynamics.Kernel.lazy_stationary_iff,Dynamics.Kernel.lazy_weight_self_ge_half,Dynamics.Kernel.lazy_selfLoop_pos,Dynamics.Kernel.lazy_weight_pos,Dynamics.Kernel.uniformNeighbor,Dynamics.Kernel.uniformNeighbor_support,Dynamics.Kernel.uniformNeighbor_expect,Dynamics.Kernel.uniformNeighbor_expect_eq_avg,Dynamics.Kernel.lazyNeighbor,Dynamics.Kernel.lazyNeighbor_weight,Dynamics.Kernel.lazyNeighbor_expect}\leanok
\uses{def:kernel,lem:neighborround}
The lazy version of a kernel $K$ is $(I+K)/2$: it puts weight at least $1/2$ on every
self-loop, charges everything $K$ charges, and has the same stationary distributions as $K$.
On a graph without isolated vertices the simple random walk $D^{-1}A$ moves to a uniform
neighbour, so its expectations are neighbour averages, and the lazy random walk is
$(I+D^{-1}A)/2$.
\end{definition}
\begin{lemma}[Sequential pairwise scheduler]\label{lem:edgeround-top}
\lean{Dynamics.oneWayStep,Dynamics.edgeRoundTopEquiv,Dynamics.card_edgeRound_top,Dynamics.sum_edgeRound_top}\leanok
\uses{lem:edgeround,lem:dartsums}
On the complete graph
the darts are the ordered pairs of distinct vertices, $n(n-1)$ of them, and a sum over them is
the sum over all ordered pairs minus the diagonal. A one-way protocol step on the dart $(u,v)$
updates the state of the responder $v$ against the initiator $u$.
\end{lemma}
\begin{proof}\leanok
A dart is an adjacent ordered pair; the complete graph has $\binom n2$ edges, each carrying two
darts.
\end{proof}
\section{Hitting times}\label{sec:dgmss-hitting}
This section formalizes Claim~2.9 of Doerr, Goldberg, Minder, Sauerwald and Scheideler,
\emph{Stabilizing consensus with the power of two choices} (SPAA 2011), as
Theorem~\ref{thm:drifthitting}. Compared with the claim, the hypothesis $c_4\log q\le q$ is added
(the target is a possible value, which a non-asymptotic statement needs), and $c_5$ also depends
on $c_1,c_2,c_3$
(\href{https://github.com/formal-dynamics/leanamics/blob/main/dynamics/FORMALIZATION_DIFFERENCES.md#hitting-time-lemma-drifthitting-maj-8-doerr-et-al-spaa-2011-claim-29}{details}).
\begin{definition}[Hitting probability]\label{def:hitprob}
\lean{Dynamics.Kernel.hitProb}\leanok\uses{def:kernel}
$\mathrm{hitProb}(K,B,n)(a)=\Pr_a(\exists s\le n,\ X_s\in B)=\Pr_a(T_B\le n)$,
the expectation over the first $n$ transitions of the indicator that one of
the states at the times $0,\dots,n$ lies in $B$.
\end{definition}
\begin{lemma}[Geometric drift bound for hitting times]\label{lem:hitdrift}
\lean{Dynamics.Kernel.stopped,Dynamics.Kernel.one_sub_hitProb_eq_event_stopped,Dynamics.Kernel.one_sub_hitProb_le_of_drift}\leanok
\uses{def:hitprob,thm:multdrift}
Let $V\ge0$ vanish on $B$ and be at least $V_{\min}>0$ off $B$, with
$\E[V(X_{t+1})\mid X_t=x]\le\rho V(x)$ for $x\notin B$. Then
$\Pr_a(T_B>t)\le\rho^tV(a)/V_{\min}$.
\end{lemma}
\begin{proof}\leanok
In the chain stopped on entering $B$, being outside $B$ at time $t$ is the
event $T_B>t$; apply Theorem~\ref{thm:multdrift} to the stopped chain with
$\delta=1-\rho$.
\end{proof}
\begin{theorem}[Hitting-time bound; Doerr, Goldberg, Minder, Sauerwald, Scheideler, SPAA 2011, Claim 2.9]\label{thm:drifthitting}
\lean{Dynamics.Kernel.drift_hitting,Dynamics.Kernel.drift_hitting_log}\leanok
\uses{lem:hitdrift}
Let $c_1>1$ and $c_2,c_3,c_4,c_6>0$. There is $c_5>0$ such that for every
finite chain and observable $X\in\{0,\dots,q\}$ with $c_4\log q\le q$: if
from every state below the target $\Pr[X_{t+1}\ge\min\{c_1X_t,q\}]\ge
1-e^{-c_2X_t}$, and $\Pr[X_{t+1}\ge1]\ge c_3$ when $X_t=0$, then
$X$ reaches $c_4\log q$ within $c_5\log q+\log_{c_1}(c_4\log q)$ steps with
probability at least $1-q^{-c_6}$ (equivalently within $C\log q$ steps).
\end{theorem}
\begin{proof}\leanok
The paper omits the proof. Let $p=\min\{c_3,1-e^{-c_2}\}$, $\theta=1/p$ and
$\beta=c_2/2$. For a large threshold $x_0$ take
$g(x)=1-\eta(\theta^x-1)$ for $x\le x_0$, with $\eta$ chosen so that
$g(x_0)=1/2$, and $g(x)=\tfrac12e^{-\beta(x-x_0)}$ for $x\ge x_0$; let
$V=g(X)$ below the target and $V=0$ on it. Below $x_0$ a step goes up by at
least one with probability at least $p$, and
$pg(x+1)+1-p\le\rho g(x)$ with $\rho=1-(1-p)\eta/2$; above $x_0$ growth by
$c_1$ fails with probability at most $e^{-c_2x}$, and the expected potential is
at most $\tfrac38e^{-\beta(x-x_0)}\le\rho g(x)$. Since $V\le1$ and
$V\ge\tfrac12q^{-\beta c_4}$ below the target, Lemma~\ref{lem:hitdrift} gives
$\Pr(T>t)\le2\rho^tq^{\beta c_4}\le q^{-c_6}$ for
$t\ge(1+\beta c_4+c_6)\log q/\log(1/\rho)$.
\end{proof}

\section{Relations between chains}\label{sec:relations}
The cross-field dictionary of Leanamics relates processes of different fields by a few kinds of
relation. Four of them concern one step of two finite kernels and are defined here; the other
kinds (the same continuous-time chain, a pathwise coupling, a large-population limit, the same
deterministic recursion) are not.
\begin{lemma}[Laws at finite times]\label{lem:law}
\lean{Dynamics.Distribution.ext,Dynamics.Distribution.weight_eq_expect_ite,Dynamics.Distribution.ext_expect,Dynamics.Distribution.map_map,Dynamics.Distribution.map_point,Dynamics.Distribution.expect_comm,Dynamics.Kernel.advance_expect,Dynamics.Kernel.law_expect}\leanok\uses{def:kernel,thm:stationary}
Distributions with the same weights, or with the same expectations, are equal. The law $pK^t$ at
time $t$ satisfies $\E_{pK^t}[g]=\E_p[K^tg]$.
\end{lemma}
\begin{proof}\leanok
Take the indicator of a point as observable; induction on $t$, exchanging two finite sums.
\end{proof}
\begin{definition}[Same chain on counts: lumping]\label{def:lumps}
\lean{Dynamics.Kernel.Lumps,Dynamics.Kernel.lumps_iff_apply}\leanok\uses{def:kernel}
A kernel $K$ on $\alpha$ lumps to a kernel $L$ on $\beta$ along $f:\alpha\to\beta$ if
$K(x)\circ f^{-1}=L(f(x))$ for every $x$, equivalently $\E_x[g(f(X_1))]=\E_{f(x)}[g(Y_1)]$ for
every observable $g$.
\end{definition}
\begin{lemma}[A lumping holds at every time]\label{lem:lumps}
\lean{Dynamics.Kernel.Lumps.apply,Dynamics.Kernel.Lumps.iterate,Dynamics.Kernel.Lumps.law,Dynamics.Kernel.Lumps.event,Dynamics.Kernel.Lumps.trajectory,Dynamics.Kernel.Lumps.hitProb,Dynamics.Kernel.Lumps.comp}\leanok\uses{def:lumps,lem:law,def:hitprob}
If $K$ lumps to $L$ along $f$, then $(pK^t)\circ f^{-1}=(p\circ f^{-1})L^t$ for every initial law
$p$ and time $t$, and the expectation of every functional of the trajectory
$f(X_0),\dots,f(X_t)$ is that of $Y_0,\dots,Y_t$ from $f(x)$. In particular the probabilities of
events at time $t$ and the hitting probabilities agree. Lumpings compose.
\end{lemma}
\begin{proof}\leanok
Induction on $t$: one step of $K$ seen through $f$ is one step of $L$.
\end{proof}
\begin{definition}[Jump chain]\label{def:jumpchain}
\lean{Dynamics.Kernel.jump,Dynamics.Kernel.SameJumpChain}\leanok\uses{def:kernel}
The jump chain $J$ of $K$ is the step conditioned on moving: $J(x,y)=K(x,y)/(1-K(x,x))$ for
$y\ne x$ and $J(x,x)=0$, except that a state with $K(x,x)=1$ stays absorbing. Two kernels have
the same jump chain if their jump chains are equal.
\end{definition}
\begin{lemma}[Kernels with the same jump chain]\label{lem:jumpchain}
\lean{Dynamics.Kernel.sum_erase_weight,Dynamics.Kernel.weight_eq_zero_of_weight_self,Dynamics.Kernel.eq_point_of_weight_self,Dynamics.Kernel.jump_of_weight_self,Dynamics.Kernel.jump_weight_self,Dynamics.Kernel.weight_eq_mul_jump,Dynamics.Kernel.apply_eq_jump,Dynamics.Kernel.apply_eq_self_iff_jump,Dynamics.Kernel.sameJumpChain_of_weight,Dynamics.Kernel.jump_jump,Dynamics.Kernel.jump_eq_self,Dynamics.Kernel.lazy_sameJumpChain,Dynamics.Kernel.SameJumpChain.symm,Dynamics.Kernel.SameJumpChain.apply_eq_self_iff}\leanok\uses{def:jumpchain,def:randomwalk}
$Kg(x)=K(x,x)g(x)+(1-K(x,x))Jg(x)$, so $g$ is harmonic at $x$ for $K$ iff it is for $J$. If the
rows of $K$ and $L$ are proportional off the diagonal with positive factors, they have the same
jump chain; a lazy kernel $(I+K)/2$ has the jump chain of $K$, and $J$ is its own jump chain.
\end{lemma}
\begin{proof}\leanok
Off the diagonal the weights of $K(x)$ add up to $1-K(x,x)$; normalize.
\end{proof}
\begin{theorem}[Same jump chain, same absorption probabilities]\label{thm:jump-absorption}
\lean{Dynamics.Kernel.event_monotone_of_absorbing,Dynamics.Kernel.apply_iSup_event,Dynamics.Kernel.event_le_of_harmonic,Dynamics.Kernel.SameJumpChain.iSup_event_eq}\leanok\uses{lem:jumpchain,thm:optional-stopping}
If $K$ and $L$ have the same jump chain and $A$ is absorbing for both, then
$\sup_t\Pr^K_x(X_t\in A)=\sup_t\Pr^L_x(Y_t\in A)$ for every $x$.
\end{theorem}
\begin{proof}\leanok
The absorption probability $h_K$ is the increasing limit of $K^t1_A$, so $Kh_K=h_K$ (a finite
sum commutes with the limit); by the previous lemma $Lh_K=h_K$. A nonnegative $L$-harmonic
function that is at least $1$ on $A$ bounds $L^t1_A$ for every $t$, by induction; hence
$h_L\le h_K$, and symmetrically.
\end{proof}
\begin{definition}[Duality]\label{def:dual}
\lean{Dynamics.Kernel.Dual,Dynamics.Kernel.Dual.symm}\leanok\uses{def:kernel}
Kernels $K$ on $\alpha$ and $L$ on $\beta$ are dual with respect to $H:\alpha\times\beta\to\mathbb R$
if $\E_x[H(X_1,y)]=\E_y[H(x,Y_1)]$ for all $x$ and $y$.
\end{definition}
\begin{lemma}[Duality at every time]\label{lem:dual}
\lean{Dynamics.Kernel.expect_iterate_comm,Dynamics.Kernel.Dual.iterate}\leanok\uses{def:dual,lem:law}
If $K$ and $L$ are dual with respect to $H$, then $\E_x[H(X_t,y)]=\E_y[H(x,Y_t)]$ for every $t$.
\end{lemma}
\begin{proof}\leanok
Induction on $t$: condition on the first step of $K$, apply the induction hypothesis, exchange the
two finite expectations and apply the one-step duality inside.
\end{proof}
\begin{definition}[Mean field]\label{def:meanfield}
\lean{Dynamics.Kernel.MeanField,Dynamics.Kernel.Lumps.meanField,Dynamics.Kernel.MeanField.iterate_id}\leanok\uses{def:lumps}
The observables $\varphi_i$ of $K$ have mean-field map $F$ if $\E_x[\varphi(X_1)]=F(\varphi(x))$
for every $x$. A lumping transfers a mean-field map, and with $F$ the identity every $\varphi_i$
keeps its expectation at all times.
\end{definition}
\begin{lemma}[The binomial law]\label{lem:binomial}
\lean{Dynamics.countFin,Dynamics.Distribution.binomial,Dynamics.Distribution.binomial_weight_eq,Dynamics.Distribution.map_decide_eq_bernoulli,Dynamics.Distribution.independent_map_pi,Dynamics.Distribution.card_countFin_eq,Dynamics.Distribution.independent_bernoulli_map_countFin}\leanok\uses{def:distribution,lem:weighted-product,lem:bridge-pmf}
The binomial law on $\{0,\dots,n\}$ has weights $\binom nk p^k(1-p)^{n-k}$, those of Mathlib's
binomial distribution. The number of heads of independent Bernoulli($p$) coins indexed by a set
of size $n$ has this law. The indicator of an event is a Bernoulli coin with the probability of
the event, and an independent product transported coordinatewise is the product of the
transported coordinates.
\end{lemma}
\begin{proof}\leanok
A configuration with $k$ heads has weight $p^k(1-p)^{n-k}$, and $\binom nk$ configurations have
$k$ heads (they correspond to the subsets of size $k$); the total mass is $(p+1-p)^n$.
\end{proof}
